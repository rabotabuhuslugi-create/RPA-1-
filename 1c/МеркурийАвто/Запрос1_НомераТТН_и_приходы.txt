// ЗАПРОС 1. Номера ТТН из Меркурия против номеров в "Приобретении товаров и услуг".
// Для каждой накладной из очереди "К оформлению" показывает приходы ТОГО ЖЕ поставщика
// с датой входящего документа в пределах +-3 дня, и все номера, которые есть в приходе
// (входящий номер, номер счета-фактуры/УПД, наш номер).
// Дату начала (01.10.2026) при необходимости поменяйте в двух местах.
// Вставьте в консоль запросов целиком и выполните. Результат выгрузите в Excel и пришлите.
// ============================================================
ВЫБРАТЬ
	ВСДок.ГрузоотправительХозяйствующийСубъект.Контрагент КАК Поставщик,
	ВСДок.ГрузоотправительХозяйствующийСубъект.Наименование КАК ПоставщикВетИС,
	ВСДок.НомерТТН КАК НомерТТН,
	ВСДок.СерияТТН КАК СерияТТН,
	НАЧАЛОПЕРИОДА(ВСДок.ДатаТТН, ДЕНЬ) КАК ДатаТТН,
	КОЛИЧЕСТВО(*) КАК ВСД,
	СУММА(ВСДок.КоличествоВЕТИС) КАК КоличествоВСД
ПОМЕСТИТЬ ТТН
ИЗ
	РегистрСведений.СтатусыОформленияДокументовВЕТИС КАК С
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.ВетеринарноСопроводительныйДокументВЕТИС КАК ВСДок
		ПО С.Основание = ВСДок.Ссылка
ГДЕ
	С.СтатусОформления = ЗНАЧЕНИЕ(Перечисление.СтатусыОформленияДокументовГосИС.НеОформлено)
	И НЕ С.Архивный
	И НЕ ВСДок.Закрыт
	И ВСДок.Дата >= ДАТАВРЕМЯ(2026, 10, 1)
СГРУППИРОВАТЬ ПО
	ВСДок.ГрузоотправительХозяйствующийСубъект.Контрагент,
	ВСДок.ГрузоотправительХозяйствующийСубъект.Наименование,
	ВСДок.НомерТТН,
	ВСДок.СерияТТН,
	НАЧАЛОПЕРИОДА(ВСДок.ДатаТТН, ДЕНЬ)
;

////////////////////////////////////////////////////////////
ВЫБРАТЬ
	Д.Ссылка КАК Ссылка,
	Д.Контрагент КАК Контрагент,
	Д.Номер КАК НашНомер,
	Д.НомерВходящегоДокумента КАК НомерВходящего,
	НАЧАЛОПЕРИОДА(Д.ДатаВходящегоДокумента, ДЕНЬ) КАК ДатаВходящего,
	Д.НомерСчетаФактуры КАК НомерСФ,
	НАЧАЛОПЕРИОДА(Д.ДатаСчетаФактуры, ДЕНЬ) КАК ДатаСФ
ПОМЕСТИТЬ Приходы
ИЗ
	Документ.ПриобретениеТоваровУслуг КАК Д
ГДЕ
	Д.Проведен
	И Д.ДатаВходящегоДокумента >= ДАТАВРЕМЯ(2026, 9, 25)
;

////////////////////////////////////////////////////////////
ВЫБРАТЬ
	Т.ПоставщикВетИС КАК ПоставщикВетИС,
	Т.НомерТТН КАК НомерТТН_Меркурий,
	Т.ДатаТТН КАК ДатаТТН,
	Т.ВСД КАК ВСД,
	Т.КоличествоВСД КАК КоличествоВСД,
	П.Ссылка КАК Приход,
	П.НомерВходящего КАК НомерВходящего_1С,
	П.ДатаВходящего КАК ДатаВходящего_1С,
	П.НомерСФ КАК НомерСФ_1С,
	П.ДатаСФ КАК ДатаСФ_1С,
	ВЫБОР
		КОГДА П.Ссылка ЕСТЬ NULL
			ТОГДА "нет прихода у поставщика за эти даты"
		КОГДА Т.НомерТТН = П.НомерВходящего
			ТОГДА "номер совпал"
		ИНАЧЕ "номер отличается"
	КОНЕЦ КАК Вывод
ИЗ
	ТТН КАК Т
		ЛЕВОЕ СОЕДИНЕНИЕ Приходы КАК П
		ПО Т.Поставщик = П.Контрагент
			И П.ДатаВходящего МЕЖДУ ДОБАВИТЬКДАТЕ(Т.ДатаТТН, ДЕНЬ, -3) И ДОБАВИТЬКДАТЕ(Т.ДатаТТН, ДЕНЬ, 3)

УПОРЯДОЧИТЬ ПО
	Т.ПоставщикВетИС,
	Т.ДатаТТН,
	Т.НомерТТН
