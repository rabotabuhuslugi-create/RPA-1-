// сколько продукций из ВСД накладной нашлось в строках прихода того же поставщика
// (через регистр "Соответствие номенклатуры ВетИС"), даты +-3 дня.
// Если у накладной с "другим" номером тут один явный приход с большим числом совпавших
// продукций - значит, сопоставлять можно по поставщику + дате + составу товаров.
// ============================================================
ВЫБРАТЬ
	ВСДок.ГрузоотправительХозяйствующийСубъект.Контрагент КАК Поставщик,
	ВСДок.ГрузоотправительХозяйствующийСубъект.Наименование КАК ПоставщикВетИС,
	ВСДок.НомерТТН КАК НомерТТН,
	НАЧАЛОПЕРИОДА(ВСДок.ДатаТТН, ДЕНЬ) КАК ДатаТТН,
	ВСДок.Продукция КАК Продукция
ПОМЕСТИТЬ ВСДСтроки
ИЗ
	РегистрСведений.СтатусыОформленияДокументовВЕТИС КАК С
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.ВетеринарноСопроводительныйДокументВЕТИС КАК ВСДок
		ПО С.Основание = ВСДок.Ссылка
ГДЕ
	С.СтатусОформления = ЗНАЧЕНИЕ(Перечисление.СтатусыОформленияДокументовГосИС.НеОформлено)
	И НЕ С.Архивный
	И НЕ ВСДок.Закрыт
	И ВСДок.Дата >= ДАТАВРЕМЯ(2026, 10, 1)
;

////////////////////////////////////////////////////////////
ВЫБРАТЬ
	Д.Ссылка КАК Ссылка,
	Д.Контрагент КАК Контрагент,
	Д.НомерВходящегоДокумента КАК Номер,
	НАЧАЛОПЕРИОДА(Д.ДатаВходящегоДокумента, ДЕНЬ) КАК ДатаВходящего,
	Т.Номенклатура КАК Номенклатура
ПОМЕСТИТЬ ПриходСтроки
ИЗ
	Документ.ПриобретениеТоваровУслуг.Товары КАК Т
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Документ.ПриобретениеТоваровУслуг КАК Д
		ПО Т.Ссылка = Д.Ссылка
ГДЕ
	Д.Проведен
	И Д.ДатаВходящегоДокумента >= ДАТАВРЕМЯ(2026, 9, 25)
;

ВЫБРАТЬ
	ВСДок.ПоставщикВетИС КАК ПоставщикВетИС,
	ВСДок.НомерТТН КАК НомерТТН_Меркурий,
	ВСДок.ДатаТТН КАК ДатаТТН,
	П.Ссылка КАК Приход,
	П.Номер КАК НомерВходящего_1С,
	КОЛИЧЕСТВО(РАЗЛИЧНЫЕ ВСДок.Продукция) КАК СовпавшихПродукций
ИЗ
	ВСДСтроки КАК ВСДок
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ РегистрСведений.СоответствиеНоменклатурыВЕТИС КАК М
		ПО ВСДок.Продукция = М.Продукция
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ ПриходСтроки КАК П
		ПО М.Номенклатура = П.Номенклатура
			И П.Контрагент = ВСДок.Поставщик
			И П.ДатаВходящего МЕЖДУ ДОБАВИТЬКДАТЕ(ВСДок.ДатаТТН, ДЕНЬ, -3) И ДОБАВИТЬКДАТЕ(ВСДок.ДатаТТН, ДЕНЬ, 3)
СГРУППИРОВАТЬ ПО
	ВСДок.ПоставщикВетИС,
	ВСДок.НомерТТН,
	ВСДок.ДатаТТН,
	П.Ссылка,
	П.Номер
УПОРЯДОЧИТЬ ПО

	ВСДок.ДатаТТН,
	ВСДок.НомерТТН
